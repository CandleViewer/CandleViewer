const RES = {};
function resolveIn(setName, name, depth) { // a token's value in a theme stack, following {references} (raw `value`s only)
  const key = setName + "|" + name;
  if (key in RES) return RES[key];

function hexOf(v) { // "#rgb", "#rrggbb", "#rrggbbaa", "rgb(...)", "rgba(...)" -> { hex: "#RRGGBB", a } or null
  if (typeof v !== "string") return null;
  const s = v.trim();
  let m = /^#([0-9a-f]{3})$/i.exec(s);
  if (m) { const c = m[1]; return { hex: ("#" + c[0] + c[0] + c[1] + c[1] + c[2] + c[2]).toUpperCase(), a: 1 }; }
  m = /^#([0-9a-f]{6})([0-9a-f]{2})?$/i.exec(s);
  if (m) return { hex: ("#" + m[1]).toUpperCase(), a: m[2] ? parseInt(m[2], 16) / 255 : 1 };
  m = /^rgba?\(\s*(\d+)[\s,]+(\d+)[\s,]+(\d+)(?:[\s,/]+([\d.]+)(%?))?\s*\)$/i.exec(s);
  if (m) {
    const h = (n) => ("0" + Math.max(0, Math.min(255, Math.round(+n))).toString(16)).slice(-2);
    const a = m[4] == null ? 1 : (m[5] ? +m[4] / 100 : +m[4]);
    return { hex: ("#" + h(m[1]) + h(m[2]) + h(m[3])).toUpperCase(), a };
  }
  return null;
}

function bindToken(shape, setName, tokenName, props) {
  try {
    const rec = { t: tokenName, s: setName, b: 0 }; // token, set, bound (1) or recorded only (0); w = why not bound
    const raw = rawOf(setName, tokenName);
    if (!raw) {
      BIND.missing++; ABSENT.set(tokenName, (ABSENT.get(tokenName) || 0) + 1);
      rec.w = "absent"; // not in the file's sets of this theme stack
      record(shape, props, rec);
      return false;
    }
    let live = BIND_MODE === "all";
    if (BIND_MODE === "active") {
      const want = resolveIn(setName, tokenName);
      const isColour = !!hexOf(want);
      live = isColour ? sameColour(activeValue(raw.tok), want) : themeInvariant(tokenName);
      if (!live) rec.w = "inactive"; // by name, the active sets would resolve another value
    }
    if (live) {
      const key = setName + "|" + tokenName + "|" + props.join(",");
      let q = QUEUE.get(key);
      if (!q) { q = { tok: raw.tok, props: props.slice(), shapes: [] }; QUEUE.set(key, q); }
      q.shapes.push(shape);
      rec.b = 1; BIND.live++;
    } else BIND.recorded++;
    record(shape, props, rec);
    return rec.b === 1;
  } catch (e) { BIND.failed++; warn("bindToken " + tokenName, e); return false; }
}
function flushBindings() { // batched application (asynchronous in Penpot); then the binding records into plugin data
  for (const q of QUEUE.values()) {

}

let pathFallback = false; // set once createPath/content fails, so every later line draws a thin rect without retrying
function mkPath(parent, content, o) { // one path from an absolute-coordinate content string; null => the caller falls back
  o = o || {};
  if (pathFallback || typeof penpot.createPath !== "function") { pathFallback = true; return null; }
  let p = null;
  try {
    p = penpot.createPath();
    p.name = TICKET + " / " + (o.name || "line");
    if ("d" in p) p.d = content; else p.content = content; // Path.d (Path.content is deprecated)
    if (!(p.width > 0 || p.height > 0)) throw new Error("path content not applied");
    if (o.fill) paintFill(p, o.fill); else p.fills = [];
    if (o.stroke) paintStroke(p, o.stroke, o.width || 1, o.dashed ? "dashed" : "solid", "center"); else p.strokes = [];

    return p;
  } catch (e) {

  }
}

function wrapCount(str, w, size, bold, mono) { // greedy word wrap with the real advances (bold + 2 %)
  const k = bold ? 1.02 : 1, wt = bold ? 600 : 400, sp = (tw(" ", size, mono, wt) - 4) * k;
  let lines = 0;
  for (const para of String(str).split("\n")) {
    let cur = 0;
    lines++;
    for (const word of para.split(" ")) { const ww = (tw(word, size, mono, wt) - 4) * k; if (cur > 0 && cur + sp + ww > w) { lines++; cur = ww; } else cur += (cur > 0 ? sp : 0) + ww; }
  }
  return lines;
}
function mockLines(str, w, size, mono) { // the line count estH and the geometry mock assume (0.56 / 0.62 em per character)

  for (const para of String(str).split("\n")) n += Math.max(1, Math.ceil(para.length / cpl));
  return n;
}

function frameFlow(f, fx, fy, fw, fh, fl) {
  const W = 400, H = 230, box2 = {};
  for (const nd of fl.nodes) {
    const b = { x: fx + nd._x, y: fy + nd._y, w: W, h: H };
    box2[nd._k] = b;
    mkRect(f, b.x, b.y, b.w, b.h, "overlay", { stroke: "borderDefault", radius: 6, radiusTok: "radius.card", name: "flow-step" });
    let cy = b.y + 14;

    for (const L of nd.lines) { mkText(f, L, b.x + 16, cy, W - 32, { size: 12, lh: 1.3, role: "text2", grow: "h", tight: true, name: "flow-step-line" }); cy += blockH(L, W - 32, 12, 1.3) + 6; }
    if (cy > b.y + H) warn("flow", nd._k + " runs " + Math.ceil(cy - b.y - H) + " px past its box");
    anchor(nd._k, b.x + b.w - 22, b.y + 22); // inside the step, right of its title
  }
  const lab = (s, x, y, w) => mkText(f, s, x, y, w, { size: 11, lh: 1.3, role: "text2", grow: "h", tight: true, name: "flow-edge-label" });

  for (const e of fl.edges) {
    const s = box2[e._f], t = box2[e._t], key = e._f + ">" + e._t;
    if (key === "open>live" || key === "live>revision" || key === "close>grace") { // horizontal, label above the line in the gutter
      const y = s.y + s.h / 2, x1 = s.x + s.w, x2 = t.x, lw = x2 - x1 - 16;
      edge([[x1, y], [x2, y]], "r");
      lab(e.label, x1 + 8, y - 8 - blockH(e.label, lw, 11, 1.3), lw);
    } else if (key === "open>retry" || key === "live>close") { // down, label right of the line
      const x = s.x + 100, y1 = s.y + s.h, y2 = t.y;
      edge([[x, y1], [x, y2]], "d");
      lab(e.label, x + 10, y1 + 12, 160);
    } else if (key === "retry>open") { // back up
      const x = s.x + 300, y1 = s.y, y2 = t.y + t.h;
      edge([[x, y1], [x, y2]], "u");
      lab(e.label, x + 10, y2 + 40, tw(e.label, 11) + 4);
    } else if (key === "grace>live") { // up and left, into live's bottom (a reopen within 30 s)
      const x1 = s.x + s.w / 2, ym = s.y - 110, x2 = t.x + 300, y2 = t.y + t.h;
      edge([[x1, s.y], [x1, ym], [x2, ym], [x2, y2]], "u");
      lab(e.label, x2 + 10, ym - 8 - blockH(e.label, x1 - x2 - 20, 11, 1.3), x1 - x2 - 20);
    } else warn("flow", "no route for " + key);
  }
}

const COL_W = 328, FACETS = D.struct.facetPrefix.map((s) => s.replace(": ", ""));
function facetLines(m) { // one line per facet the sources give; the facets they do not give share one line, each with "—"
  const out = [], none = [];
  for (const fct of FACETS) { const v = m[fct] == null ? DASH : m[fct]; if (v === DASH) { none.push(fct + ": " + DASH); continue; } for (const s of Array.isArray(v) ? v : [v]) out.push(fct + ": " + s); }
  if (none.length) out.push(none.join(" · "));
  return out;
}

function markerBlock(ctx, m, no, x, y, W, keys) {
  const pad = 10, innerX = x + 36, innerW = W - 46;
  let h = pad;
  const lines = [{ s: m.part, o: { size: 11.5, weight: 700, name: "marker-part", after: 3, role: "text" } }];
  if (m.note) lines.push({ s: m.note, o: { size: 11, role: "text3", name: "marker-note", after: 3 } });
  for (const fl of facetLines(m)) lines.push({ s: fl, o: { size: 11, role: "text2", name: "marker-facet", after: 2 } });
  let bg = null;
  if (ctx.draw) {
    bg = mkRect(ctx.b, x, y, W, 10, "raised", Object.assign({ stroke: "borderSubtle", radius: 6, role: "card-bg", bg: true, pad, gap: 12, name: "marker-block" }, keys));
    ctx.shapes.push(bg);
    badge(ctx, x + 9, y + 9, no, keys);
  }
  for (const L of lines) h += ctext(ctx, L.s, innerX, y + h, innerW, Object.assign({}, keys, L.o)) + L.o.after;

  if (bg) bg.resize(W, h);

}

function annotate(b, bd, colX, colY, sfx) { // markers on the parts + the column, grouped
